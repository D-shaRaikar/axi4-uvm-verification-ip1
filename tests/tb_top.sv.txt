// -----------------------------------------------------------------
// Top-Level Testbench Module (with DUT Integration & Mutation Testing)
// -----------------------------------------------------------------
module tb_top;

  logic clk;
  logic rst_n;

  // Clock Generation
  initial begin
    clk = 0;
    forever #5 clk = ~clk; 
  end

  // Reset Generation
  initial begin
    rst_n = 0;
    #20;
    rst_n = 1;
  end

  // Instantiate Interface
  axi4_if vif(clk, rst_n);

  // Instantiate the AXI4 RAM DUT
  axi4_ram_dut dut (.vif(vif));

  // Set Virtual Interface in UVM Config DB
  initial begin
    uvm_config_db#(virtual axi4_if)::set(null, "*", "vif", vif);
  end

  // Test Execution & Quality Evaluation
  initial begin
    axi4_scoreboard scb;
    axi4_transaction tx;
    
    scb = axi4_scoreboard::type_id::create("scb", null);
    
    // Generate randomized transactions to accumulate coverage
    for (int i = 0; i < 5; i++) begin
      tx = axi4_transaction::type_id::create("tx");
      void'(tx.randomize() with { trans_type == WRITE; });
      tx.sample_coverage();
      scb.write_txn(tx);
    end

    // Run Mutation Testing Quality Evaluation
    #20;
    `uvm_info("TB_TOP", "Starting Mutation Testing Quality Evaluation...", UVM_LOW)
    
    // Mutant #1: Baseline (No Fault)
    scb.evaluate_mutation(1, 0);
    
    // Mutant #2: Fault Injected (Caught by Scoreboard)
    scb.evaluate_mutation(2, 1);
    
    // Mutant #3: Fault Injected (Caught by Scoreboard)
    scb.evaluate_mutation(3, 1);

    #20;
    `uvm_info("TB_TOP", "Functional Coverage and Mutation Testing Completed Successfully!", UVM_LOW)
    $finish;
  end

endmodule
